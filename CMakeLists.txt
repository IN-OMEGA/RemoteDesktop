cmake_minimum_required(VERSION 3.24)
project(RemoteCore VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "RemoteCore initial executable requires Windows")
endif()
add_executable(RemoteCore WIN32 src/main.cpp src/Diagnostics/SystemInfo.cpp src/ScreenCapture/DesktopDuplication.cpp)
target_compile_features(RemoteCore PRIVATE cxx_std_20)
target_compile_definitions(RemoteCore PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(RemoteCore PRIVATE user32 dxgi d3d11)
add_library(RemoteCoreCapture SHARED src/ScreenCapture/Bridge.cpp)
target_compile_features(RemoteCoreCapture PRIVATE cxx_std_20)
target_compile_definitions(RemoteCoreCapture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(RemoteCoreCapture PRIVATE d3d11 dxgi user32 d3dcompiler)
if(MSVC)
  set_property(TARGET RemoteCore RemoteCoreCapture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(RemoteCore PRIVATE /utf-8)
  target_compile_options(RemoteCoreCapture PRIVATE /utf-8 /W4)
endif()
